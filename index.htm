function confirmarProd(id, catNombre, numOpc, opcionesStr) {

  var p = PRODUCTOS.find(function(pr){ return pr.id == id; }); if (!p) return;

  var qty = parseInt(document.getElementById('qty-'+id).textContent) || 1;

  var opcsElegidas = [];

  if (numOpc > 0 && opcionesStr) {

    var opts = opcionesStr.split(',').map(function(o){ return o.trim(); }).filter(Boolean);

    opts.forEach(function(opt, i) {

      var el = document.getElementById('opc-'+id+'-'+i);

      if (el && el.value.trim()) opcsElegidas.push(opt+': '+el.value.trim());

    });

  }

  // Cada UNIDAD se agrega como línea independiente del carrito (nunca se agrupan

  // en una sola con cantidad>1), para que en 'Datos de envío por producto' se

  // pueda capturar un destinatario, dirección y dedicatoria distintos por cada

  // unidad, aunque sea el mismo producto pedido varias veces.

  var precioFinal = (p.tieneDescuento && p.precioConDesc > 0) ? p.precioConDesc : p.precio;

  for (var k = 0; k < qty; k++) {

    carrito.push({ _prodId:id, tipo:catNombre, descripcion:p.descripcion,

      precio: precioFinal>0 ? precioFinal : 0,

      precioUnit: precioFinal, precioOriginal: p.precio,

      tieneDescuento: p.tieneDescuento || false,

      precioConDesc: p.precioConDesc || 0,

      cantidad:1,

      imagen:p.imagen, personalizacion:null, opcionesElegidas:opcsElegidas.slice() });

  }

  var card = document.getElementById('prodcard-'+id);

  if (card) card.classList.add('confirmado');

  document.getElementById('qty-'+id).textContent = '1';

  renderCarrito();

  showToast('✅ '+p.descripcion+' agregado (x'+qty+')');

}

function calcularDistancia(dir) {

  var badge = document.getElementById('distancia-badge');

  if (!badge) return;

  badge.textContent = '📏 Calculando...';

  badge.className = 'distancia-badge calculando';

  var sucCoords = SUC_COORDS[sucursal];

  if (!sucCoords) { badge.textContent = '📏 Selecciona una sucursal'; return; }

  fetch('https://nominatim.openstreetmap.org/search?format=json&limit=1&q='+encodeURIComponent(dir+', México'), {

    headers: { 'Accept-Language': 'es' }

  }).then(function(r){ return r.json(); })

  .then(function(data) {

    if (!data || !data.length) {

      badge.textContent = '📏 No se pudo calcular — verifica la dirección';

      badge.className = 'distancia-badge calculando';

      return;

    }

    var lat2 = parseFloat(data[0].lat);

    var lon2 = parseFloat(data[0].lon);

    var km = haversine(sucCoords.lat, sucCoords.lng, lat2, lon2);

    badge.textContent = '📏 Aprox. '+km.toFixed(1)+' km desde Sucursal '+sucursal;

    badge.className = 'distancia-badge';

    badge.dataset.km = km.toFixed(1);

  }).catch(function(){

    badge.textContent = '📏 No disponible en este momento';

    badge.className = 'distancia-badge calculando';

  });

}

function haversine(lat1, lon1, lat2, lon2) {

  var R = 6371;

  var dLat = (lat2-lat1)*Math.PI/180;

  var dLon = (lon2-lon1)*Math.PI/180;

  var a = Math.sin(dLat/2)*Math.sin(dLat/2)+

          Math.cos(lat1*Math.PI/180)*Math.cos(lat2*Math.PI/180)*

          Math.sin(dLon/2)*Math.sin(dLon/2);

  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));

}
